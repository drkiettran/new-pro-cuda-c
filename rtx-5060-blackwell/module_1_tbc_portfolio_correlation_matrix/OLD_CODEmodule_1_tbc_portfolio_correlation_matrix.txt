__global__ void __launch_bounds__(TILE_SIZE* TILE_SIZE)
compute_correlation_matrix_tbc(
    const float* __restrict__ asset_returns,
    float* __restrict__ correlation_matrix,
    const int n_assets,
    const int n_days
) {
    // Initialize cluster and block groups
    cg::cluster_group cluster = cg::this_cluster();
    cg::thread_block block = cg::this_thread_block();
    
    // Cluster dimensions
    const int cluster_rank = cluster.block_rank();
    const int blocks_per_cluster = cluster.num_blocks();
    
    // Thread and block indices
    const int tx = threadIdx.x;
    const int ty = threadIdx.y;
    const int block_x = blockIdx.x;
    const int block_y = blockIdx.y;
    
    // Global indices for this thread
    const int global_x = block_x * TILE_SIZE + tx;
    const int global_y = block_y * TILE_SIZE + ty;
    
    if (global_x >= n_assets || global_y >= n_assets) return;
    
    // Distributed shared memory - accessible across cluster
    extern __shared__ float shared_mem[];
    float* local_means = shared_mem;
    float* local_stds = &shared_mem[TILE_SIZE];
    float* temp_buffer = &shared_mem[2 * TILE_SIZE];
    
    // Phase 1: Compute means using cluster-wide cooperation
    if (ty == 0 && tx < TILE_SIZE && global_x < n_assets) {
        float sum = 0.0f;
        
        // Each block in cluster processes different portions
        int start_day = (cluster_rank * n_days) / blocks_per_cluster;
        int end_day = ((cluster_rank + 1) * n_days) / blocks_per_cluster;
        
        for (int day = start_day; day < end_day; ++day) {
            sum += asset_returns[global_x * n_days + day];
        }
        
        temp_buffer[tx] = sum;
    }
    
    block.sync();
    
    // Cluster-wide reduction for means
    if (ty == 0 && tx < TILE_SIZE) {
        cluster.sync();  // Synchronize all blocks in cluster
        
        // Aggregate partial sums from all blocks in cluster
        float total_sum = 0.0f;
        for (int b = 0; b < blocks_per_cluster; ++b) {
            // Access distributed shared memory from other blocks
            float* remote_buffer = cluster.map_shared_rank(temp_buffer, b);
            if (tx < TILE_SIZE) {
                total_sum += remote_buffer[tx];
            }
        }
        
        local_means[tx] = total_sum / n_days;
    }
    
    block.sync();
    
    // Phase 2: Compute standard deviations with cluster cooperation
    if (ty == 0 && tx < TILE_SIZE && global_x < n_assets) {
        float sum_sq = 0.0f;
        float mean = local_means[tx];
        
        int start_day = (cluster_rank * n_days) / blocks_per_cluster;
        int end_day = ((cluster_rank + 1) * n_days) / blocks_per_cluster;
        
        for (int day = start_day; day < end_day; ++day) {
            float diff = asset_returns[global_x * n_days + day] - mean;
            sum_sq += diff * diff;
        }
        
        temp_buffer[tx] = sum_sq;
    }
    
    block.sync();
    
    // Cluster-wide reduction for standard deviations
    if (ty == 0 && tx < TILE_SIZE) {
        cluster.sync();
        
        float total_sum_sq = 0.0f;
        for (int b = 0; b < blocks_per_cluster; ++b) {
            float* remote_buffer = cluster.map_shared_rank(temp_buffer, b);
            if (tx < TILE_SIZE) {
                total_sum_sq += remote_buffer[tx];
            }
        }
        
        local_stds[tx] = sqrtf(total_sum_sq / (n_days - 1));
    }
    
    block.sync();
    
    // Phase 3: Compute correlation using cluster resources
    if (global_x < n_assets && global_y < n_assets) {
        float mean_x = local_means[tx];
        float mean_y = local_means[ty];
        float std_x = local_stds[tx];
        float std_y = local_stds[ty];
        
        // Distribute covariance computation across cluster
        float covariance = 0.0f;
        int start_day = (cluster_rank * n_days) / blocks_per_cluster;
        int end_day = ((cluster_rank + 1) * n_days) / blocks_per_cluster;
        
        for (int day = start_day; day < end_day; ++day) {
            float ret_x = asset_returns[global_x * n_days + day] - mean_x;
            float ret_y = asset_returns[global_y * n_days + day] - mean_y;
            covariance += ret_x * ret_y;
        }
        
        // Store partial covariance
        temp_buffer[ty * TILE_SIZE + tx] = covariance;
    }
    
    block.sync();
    cluster.sync();
    
    // Final aggregation and correlation calculation
    if (global_x < n_assets && global_y < n_assets) {
        float total_covariance = 0.0f;
        
        // Aggregate covariances from all blocks in cluster
        for (int b = 0; b < blocks_per_cluster; ++b) {
            float* remote_buffer = cluster.map_shared_rank(temp_buffer, b);
            total_covariance += remote_buffer[ty * TILE_SIZE + tx];
        }
        
        total_covariance /= (n_days - 1);
        
        // Compute correlation coefficient
        float std_product = local_stds[tx] * local_stds[ty];
        float correlation;
        
        if (std_product > 1e-8f) {
            correlation = total_covariance / std_product;
        } else {
            correlation = (global_x == global_y) ? 1.0f : 0.0f;
        }
        
        correlation = fmaxf(-1.0f, fminf(1.0f, correlation));
        
        // Write to global memory
        correlation_matrix[global_x * n_assets + global_y] = correlation;
    }
}

__global__ void init_rng_kernel(curandState_t* states, int n_states) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < n_states) {
        curand_init(1234 + idx, idx, 0, &states[idx]);
    }
}

__global__ void generate_monte_carlo_paths(
    const float* __restrict__ correlation_matrix,
    curandState_t* __restrict__ rng_states,
    float* __restrict__ mc_paths,
    const int n_assets,
    const int n_paths
) {
    int path_id = blockIdx.x * blockDim.x + threadIdx.x;
    if (path_id >= n_paths) return;

    curandState_t local_state = rng_states[path_id];

    for (int asset = 0; asset < n_assets; ++asset) {
        float random_return = curand_normal(&local_state);
        float correlated_return = random_return;

        for (int j = 0; j < asset && j < 10; ++j) {
            float correlation = correlation_matrix[asset * n_assets + j];
            if (correlation > 0.1f) {
                correlated_return += correlation * 0.1f * curand_normal(&local_state);
            }
        }

        mc_paths[path_id * n_assets + asset] = correlated_return;
    }

    rng_states[path_id] = local_state;
}

__global__ void calculate_portfolio_pnl(
    const float* __restrict__ mc_paths,
    const float* __restrict__ portfolio_weights,
    float* __restrict__ portfolio_pnl,
    const int n_assets,
    const int n_paths
) {
    int path_id = blockIdx.x * blockDim.x + threadIdx.x;
    if (path_id >= n_paths) return;

    float pnl = 0.0f;
    for (int asset = 0; asset < n_assets; ++asset) {
        pnl += portfolio_weights[asset] * mc_paths[path_id * n_assets + asset];
    }

    portfolio_pnl[path_id] = pnl;
}

void generate_synthetic_market_data(PortfolioData& data) {
    std::random_device rd;
    std::mt19937 gen(rd());
    std::normal_distribution<float> return_dist(0.0f, 0.005f);
    std::uniform_real_distribution<float> weight_dist(0.0f, 1.0f);

    std::vector<float> host_returns(N_ASSETS * N_TRADING_DAYS);
    for (int asset = 0; asset < N_ASSETS; ++asset) {
        for (int day = 0; day < N_TRADING_DAYS; ++day) {
            float sector_effect = return_dist(gen) * 0.3f;
            float idiosyncratic = return_dist(gen) * 0.7f;
            host_returns[asset * N_TRADING_DAYS + day] = sector_effect + idiosyncratic;
        }
    }

    std::vector<float> host_weights(N_ASSETS);
    float weight_sum = 0.0f;
    for (int i = 0; i < N_ASSETS; ++i) {
        host_weights[i] = weight_dist(gen);
        weight_sum += host_weights[i];
    }
    for (int i = 0; i < N_ASSETS; ++i) {
        host_weights[i] /= weight_sum;
    }

    CUDA_CHECK(cudaMemcpy(data.asset_returns, host_returns.data(),
        N_ASSETS * N_TRADING_DAYS * sizeof(float), cudaMemcpyHostToDevice));
    CUDA_CHECK(cudaMemcpy(data.portfolio_weights, host_weights.data(),
        N_ASSETS * sizeof(float), cudaMemcpyHostToDevice));
}

VaRResults calculate_var(float* portfolio_pnl, int n_paths) {
    thrust::device_ptr<float> pnl_ptr(portfolio_pnl);
    thrust::sort(pnl_ptr, pnl_ptr + n_paths);

    VaRResults results;
    int idx_95 = static_cast<int>(0.05 * n_paths);
    int idx_99 = static_cast<int>(0.01 * n_paths);
    int idx_999 = static_cast<int>(0.001 * n_paths);

    float host_results[3];
    CUDA_CHECK(cudaMemcpy(&host_results[0], portfolio_pnl + idx_95, sizeof(float), cudaMemcpyDeviceToHost));
    CUDA_CHECK(cudaMemcpy(&host_results[1], portfolio_pnl + idx_99, sizeof(float), cudaMemcpyDeviceToHost));
    CUDA_CHECK(cudaMemcpy(&host_results[2], portfolio_pnl + idx_999, sizeof(float), cudaMemcpyDeviceToHost));

    results.var_95 = -host_results[0];
    results.var_99 = -host_results[1];
    results.var_999 = -host_results[2];

    float sum_tail = 0.0f;
    for (int i = 0; i < idx_95; ++i) {
        float tail_val;
        CUDA_CHECK(cudaMemcpy(&tail_val, portfolio_pnl + i, sizeof(float), cudaMemcpyDeviceToHost));
        sum_tail += tail_val;
    }
    results.cvar_95 = (idx_95 > 0) ? -sum_tail / idx_95 : 0.0f;

    return results;
}

    float* __restrict__ correlation_matrix,   // [N_ASSETS x N_ASSETS]
    const int n_assets,
    const int n_days
) {
    // Initialize cooperative groups
    auto block = cg::this_thread_block();

    // Calculate tile and thread positions
    const int tile_x = blockIdx.x;
    const int tile_y = blockIdx.y;
    const int tx = threadIdx.x;
    const int ty = threadIdx.y;

    // Global matrix indices
    const int global_x = tile_x * TILE_SIZE + tx;
    const int global_y = tile_y * TILE_SIZE + ty;

    // Bounds checking
    if (global_x >= n_assets || global_y >= n_assets) return;

    // Shared memory for statistics
    extern __shared__ float shared_mem[];
    float* means_a = shared_mem;
    float* means_b = &shared_mem[TILE_SIZE];
    float* stds_a = &shared_mem[2 * TILE_SIZE];
    float* stds_b = &shared_mem[3 * TILE_SIZE];

    // Compute means for asset X (row-wise)
    if (ty == 0 && tx < TILE_SIZE && global_x < n_assets) {
        float sum = 0.0f;
        for (int day = 0; day < n_days; ++day) {
            sum += asset_returns[global_x * n_days + day];
        }
        means_a[tx] = sum / n_days;
    }

    // Compute means for asset Y (column-wise)
    if (tx == 0 && ty < TILE_SIZE && global_y < n_assets) {
        float sum = 0.0f;
        for (int day = 0; day < n_days; ++day) {
            sum += asset_returns[global_y * n_days + day];
        }
        means_b[ty] = sum / n_days;
    }

    block.sync();

    // Compute standard deviations for asset X
    if (ty == 0 && tx < TILE_SIZE && global_x < n_assets) {
        float sum_sq = 0.0f;
        for (int day = 0; day < n_days; ++day) {
            float diff = asset_returns[global_x * n_days + day] - means_a[tx];
            sum_sq += diff * diff;
        }
        stds_a[tx] = sqrtf(sum_sq / (n_days - 1));
    }

    // Compute standard deviations for asset Y
    if (tx == 0 && ty < TILE_SIZE && global_y < n_assets) {
        float sum_sq = 0.0f;
        for (int day = 0; day < n_days; ++day) {
            float diff = asset_returns[global_y * n_days + day] - means_b[ty];
            sum_sq += diff * diff;
        }
        stds_b[ty] = sqrtf(sum_sq / (n_days - 1));
    }

    block.sync();

    // Compute correlation coefficient
    float correlation = 0.0f;
    if (global_x < n_assets && global_y < n_assets) {
        // Compute covariance
        float covariance = 0.0f;
        for (int day = 0; day < n_days; ++day) {
            float ret_x = asset_returns[global_x * n_days + day] - means_a[tx];
            float ret_y = asset_returns[global_y * n_days + day] - means_b[ty];
            covariance += ret_x * ret_y;
        }
        covariance /= (n_days - 1);

        // Compute correlation coefficient
        float std_product = stds_a[tx] * stds_b[ty];
        if (std_product > 1e-8f) {
            correlation = covariance / std_product;
        }
        else {
            correlation = (global_x == global_y) ? 1.0f : 0.0f;
        }

        // Ensure correlation is in valid range [-1, 1]
        correlation = fmaxf(-1.0f, fminf(1.0f, correlation));
    }

    // Write result to global memory
    if (global_x < n_assets && global_y < n_assets) {
        correlation_matrix[global_x * n_assets + global_y] = correlation;
    }
}

    const int n_assets,

        float correlated_return = random_return; // Start with base random return

__global__ void calculate_portfolio_pnl(

    const float* __restrict__ portfolio_weights,

    const int n_paths
) {

// VaR calculation function
VaRResults calculate_var(float* portfolio_pnl, int n_paths) {
    // Sort P&L values using Thrust
    thrust::device_ptr<float> pnl_ptr(portfolio_pnl);
    thrust::sort(pnl_ptr, pnl_ptr + n_paths);

    VaRResults results;

    // Calculate VaR at different confidence levels
    int idx_95 = static_cast<int>(0.05 * n_paths);    // 5th percentile (95% VaR)
    int idx_99 = static_cast<int>(0.01 * n_paths);    // 1st percentile (99% VaR)
    int idx_999 = static_cast<int>(0.001 * n_paths);  // 0.1th percentile (99.9% VaR)

    // Copy results to host
    float host_results[4];
    CUDA_CHECK(cudaMemcpy(&host_results[0], portfolio_pnl + idx_95, sizeof(float), cudaMemcpyDeviceToHost));
    CUDA_CHECK(cudaMemcpy(&host_results[1], portfolio_pnl + idx_99, sizeof(float), cudaMemcpyDeviceToHost));
    CUDA_CHECK(cudaMemcpy(&host_results[2], portfolio_pnl + idx_999, sizeof(float), cudaMemcpyDeviceToHost));

    results.var_95 = -host_results[0];   // VaR is positive (loss)
    results.var_99 = -host_results[1];
    results.var_999 = -host_results[2];

    // Calculate Conditional VaR (Expected Shortfall) for 95%
    float sum_tail = 0.0f;
    for (int i = 0; i < idx_95; ++i) {
        float tail_val;
        CUDA_CHECK(cudaMemcpy(&tail_val, portfolio_pnl + i, sizeof(float), cudaMemcpyDeviceToHost));
        sum_tail += tail_val;
    }
    results.cvar_95 = (idx_95 > 0) ? -sum_tail / idx_95 : 0.0f;

    return results;
}

// Optimized correlation matrix computation kernel
__global__ void compute_correlation_matrix_optimized(
    const float* __restrict__ asset_returns,  // [N_ASSETS x N_TRADING_DAYS]
    float* __restrict__ correlation_matrix,   // [N_ASSETS x N_ASSETS]
    const int n_assets,
    const int n_days

    auto block = cg::this_thread_block();

    const int tile_x = blockIdx.x;
    const int tile_y = blockIdx.y;
    const int tx = threadIdx.x;
    const int ty = threadIdx.y;

    // Global matrix indices
    const int global_x = tile_x * TILE_SIZE + tx;
    const int global_y = tile_y * TILE_SIZE + ty;

    // Bounds checking

    float* means_a = shared_mem;

        for (int day = 0; day < n_days; ++day) {
            sum += asset_returns[global_x * n_days + day];
        }
        means_a[tx] = sum / n_days;
    }

    // Compute means for asset Y (column-wise)
    if (tx == 0 && ty < TILE_SIZE && global_y < n_assets) {
        float sum = 0.0f;
        for (int day = 0; day < n_days; ++day) {
            sum += asset_returns[global_y * n_days + day];
        }
        means_b[ty] = sum / n_days;
    }

    block.sync();

    // Compute standard deviations for asset X
    if (ty == 0 && tx < TILE_SIZE && global_x < n_assets) {
        float sum_sq = 0.0f;
        for (int day = 0; day < n_days; ++day) {
            float diff = asset_returns[global_x * n_days + day] - means_a[tx];
            sum_sq += diff * diff;
        }

    if (global_x < n_assets && global_y < n_assets) {

        for (int day = 0; day < n_days; ++day) {

        correlation = fmaxf(-1.0f, fminf(1.0f, correlation));
    }

    // Write result to global memory
    if (global_x < n_assets && global_y < n_assets) {

// Monte Carlo path generation kernel

    const int n_assets,

    }

    rng_states[path_id] = local_state;

    // Sort P&L values using Thrust
    thrust::device_ptr<float> pnl_ptr(portfolio_pnl);
    thrust::sort(pnl_ptr, pnl_ptr + n_paths);

    VaRResults results;

    // Calculate VaR at different confidence levels
    int idx_95 = static_cast<int>(0.05 * n_paths);    // 5th percentile (95% VaR)
    int idx_99 = static_cast<int>(0.01 * n_paths);    // 1st percentile (99% VaR)
    int idx_999 = static_cast<int>(0.001 * n_paths);  // 0.1th percentile (99.9% VaR)

    // Copy results to host
    float host_results[4];
    CUDA_CHECK(cudaMemcpy(&host_results[0], portfolio_pnl + idx_95, sizeof(float), cudaMemcpyDeviceToHost));
    CUDA_CHECK(cudaMemcpy(&host_results[1], portfolio_pnl + idx_99, sizeof(float), cudaMemcpyDeviceToHost));
    CUDA_CHECK(cudaMemcpy(&host_results[2], portfolio_pnl + idx_999, sizeof(float), cudaMemcpyDeviceToHost));

    results.var_95 = -host_results[0];   // VaR is positive (loss)
    results.var_99 = -host_results[1];
    results.var_999 = -host_results[2];

    // Calculate Conditional VaR (Expected Shortfall) for 95%
    float sum_tail = 0.0f;
    for (int i = 0; i < idx_95; ++i) {
        float tail_val;
        CUDA_CHECK(cudaMemcpy(&tail_val, portfolio_pnl + i, sizeof(float), cudaMemcpyDeviceToHost));
        sum_tail += tail_val;
    }
    results.cvar_95 = (idx_95 > 0) ? -sum_tail / idx_95 : 0.0f;

    return results;
}
